// ❌ BEFORE (CRASHES):
// ────────────────────
~AlsaMidiSequencerInputClient() {
	delete alsa_seq_client_rx_queue;  // 💥 Threads still using it!
}

// ✓ AFTER (SAFE):
// ───────────────
~AlsaMidiSequencerInputClient() {
	// 1️⃣ Signal
	midi_in_client_thread_is_running = false;

	// 2️⃣ Wake
	if (alsa_seq_client_rx_queue) {
		alsa_seq_client_rx_queue->shutdown();  // Wake waiting threads
	}

	// 3️⃣ Wait
	if (midi_client_in_thread) {
		pthread_join(midi_client_in_thread, nullptr);  // WAIT for thread exit
	}

	// 4️⃣ Delete (NOW it's safe)
	delete alsa_seq_client_rx_queue;  // ✓ No one is using it anymore
}

 *

 * 
 *   ALWAYS destroy objects in this order:
 *   1. Stop new work (set flag)

 *   3. Wait for threads to exit (join)

 *   ❌ Delete while threads are running

 *   ❌ Delete without joining threads first
